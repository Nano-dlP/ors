# 1) Copiar el dump a /tmp (para que postgres lo pueda leer)
sudo cp "/home/nano/Escritorio/backup/Produccion/db_salud_mental_2026-03-10_11-52.sql" "/tmp/db_salud_mental_2026-03-10_11-52.sql"

# 2) Darle permisos/propietario al usuario postgres

# 3) Restaurar en la base ya creada
